    Ở Bài 3, chúng ta đã sở hữu hàm tính toán PID hoàn chỉnh với các phép tinh chỉnh toán học mượt mà. Nhưng hãy thử tưởng tượng bạn đang chế tạo một chiếc xe 4 bánh lái độc lập hoặc một gimbal 2 trục xoay: bạn sẽ cần chạy 2 đến 4 bộ điều khiển PID cùng lúc.

    Nếu tiếp tục dùng các biến toàn cục rời rạc (I_term, previous_error, previous_time…), bạn sẽ phải nhân bản hàng chục dòng biến toàn cục, gây xung đột bộ nhớ và không thể bảo trì. Đã đến lúc đưa toàn bộ thuật toán vào một Lớp đối tượng C++ (Class PIDController) chuẩn mực.

    1. Ba mảnh ghép tối ưu công nghiệp bổ sung

    Trước khi đóng gói, chúng ta trang bị thêm 3 cơ chế thực chiến rất phổ biến trong các thư viện điều khiển nhúng:

    A. Vùng chết (Deadband / DeadZone)

    Khi sai số $|e|$ đã rất nhỏ (nằm trong phạm vi sai số cho phép của cảm biến), việc cơ cấu chấp hành liên tục nhích qua lại từng phân số milimet sẽ gây rung rơ cơ khí và tốn điện vô ích. Nếu $|e| < \text{DeadBand}$, ta coi như sai số bằng 0:

    \[\begin{aligned} e_{\text{eff}} = \begin{cases} 0 & \text{khi } |e| < \text{DeadBand} \\ e & \text{khi } |e| \ge \text{DeadBand} \end{cases} \end{aligned}\]
    // Vùng chết (Deadband)
if (abs(error) < DeadBand) {
  error = 0.0f;
}

    B. Bộ lọc ngõ ra (Output Filter)

    Để bảo vệ mạch công suất và cuộn dây động cơ khỏi các cú thay đổi đột ngột từ tín hiệu điều khiển, ta đưa $Output$ qua một bộ lọc thông thấp bậc một:

    \[Output_{\text{filt}}(k) = \alpha_{\text{out}} \cdot Output(k) + (1 - \alpha_{\text{out}}) \cdot Output_{\text{filt}}(k-1)\]
    // Bộ lọc thông thấp ngõ ra
output = Output_Alpha * output + (1.0f - Output_Alpha) * previous_output;

    C. Xử lý kẹt động cơ (Error Handling & Motor Blocked)

    Nếu $Output$ đã vọt lên tối đa ($100\%$) trong thời gian dài nhưng giá trị đo $Input$ vẫn đứng yên một chỗ, chứng tỏ cơ cấu đã bị kẹt vật lý. Nếu tiếp tục cấp điện, động cơ sẽ bốc khói. Thư viện sẽ tự động kích hoạt cờ lỗi và ngắt $Output = 0$.

    // Chống kẹt động cơ (Motor Blocked)
// Xét CẢ HAI biên: động cơ chạy lùi cũng phải phát hiện được kẹt.
bool at_limit = (previous_output >= Max_Output) || (previous_output <= Min_Output);
if (at_limit && fabs(input - previous_input) < 0.05f) {
  blocked_counter++;
  if (blocked_counter > 30) { // Kẹt quá 30 chu kỳ
    is_blocked = true;
    return 0.0f; // Ngắt công suất an toàn
  }
} else {
  blocked_counter = 0;
}

    Chỉ kiểm tra previous_output >= Max_Output là một cái bẫy kinh điển: khi cơ cấu chạy chiều âm và bị chèn cứng ở $-100\%$, điều kiện đó không bao giờ đúng nên cơ chế bảo vệ im lặng không hoạt động — đúng lúc động cơ đang nóng nhất.

    2. Kiến trúc Cờ tính năng (Improve Flags Bitmask) của WangHongxi

    Mỗi dự án nhúng lại có những yêu cầu phần cứng khác nhau: có đối tượng cần lọc ngõ ra, có đối tượng chỉ cần khâu $P$ cơ bản. Thay vì viết nhiều hàm con riêng biệt, tác giả WangHongxi trong thư viện nổi tiếng WangHongxi2001/PID_Library sử dụng mặt nạ bit (Bitmask) để quản lý các tính năng cải tiến:

    // Định nghĩa các cờ tính năng bằng số nhị phân độc lập
enum PID_Improve_e {
  PID_IMPROVE_NONE               = 0x00,
  PID_IMPROVE_INTEGRAL_LIMIT     = 0x01, // 0b00000001: Giới hạn khâu I
  PID_IMPROVE_DERIV_ON_MEAS      = 0x02, // 0b00000010: Vi phân theo giá trị đo
  PID_IMPROVE_TRAPEZOID_INT      = 0x04, // 0b00000100: Tích phân hình thang
  PID_IMPROVE_CHANGING_INT       = 0x08, // 0b00001000: Biến tốc độ tích phân
  PID_IMPROVE_OUTPUT_FILTER      = 0x10, // 0b00010000: Bộ lọc đầu ra
  PID_IMPROVE_DEADBAND           = 0x20, // 0b00100000: Vùng chết sai số
  PID_IMPROVE_ERROR_HANDLE       = 0x40, // 0b01000000: Tự ngắt khi kẹt motor
  PID_IMPROVE_DERIV_FILTER       = 0x80  // 0b10000000: Lọc thông thấp khâu D
};

    Hai điểm cần cẩn thận khi đối chiếu với mã nguồn gốc.

    Thứ nhất, cách đánh số bit ở đây khác bản gốc. Ví dụ 0x08 trong bảng trên là biến tốc độ tích phân, còn trong PID.h của WangHongxi thì 0x08 là Proportional_On_Measurement và biến tốc độ tích phân nằm ở 0x20. Nếu bạn copy giá trị cờ qua lại giữa hai bên sẽ bật nhầm tính năng.

    Thứ hai, thứ tự tham số ngược nhau. Ở đây là calculate(setpoint, input), còn bản gốc là PID_Calculate(pid, measure, target) — giá trị đo đứng trước giá trị đặt. Đảo nhầm hai đối số này thì hệ đảo dấu phản hồi và chạy loạn ngay.

    3. Bản mã nguồn Thư viện PID hoàn chỉnh (PID.h)

    Dưới đây là mã nguồn C++ đóng gói toàn diện, sẵn sàng copy vào bất kỳ dự án Arduino, STM32 hay ESP32 nào:

    #pragma once
#include <Arduino.h>

class PIDController {
public:
  float Kp, Ki, Kd;
  float Min_Output, Max_Output; // dải hai chiều: Min âm cho động cơ đảo chiều
  float Max_I;
  float DeadBand;
  float ScalarA, ScalarB;
  float Output_Alpha;           // hệ số lọc ngõ ra
  float Deriv_Alpha;            // hệ số lọc khâu D
  uint8_t Improve;              // Mặt nạ bit chứa các cờ tính năng

  bool is_blocked;

  PIDController(float kp = 2.0, float ki = 0.0, float kd = 0.0,
                float min_out = -255.0, float max_out = 255.0, float max_i = 25.0) {
    Kp = kp; Ki = ki; Kd = kd;
    Min_Output = min_out; Max_Output = max_out; Max_I = max_i;
    DeadBand = 0.5f; ScalarA = 20.0f; ScalarB = 5.0f;
    Output_Alpha = 0.3f; Deriv_Alpha = 0.25f;
    Improve = PID_IMPROVE_INTEGRAL_LIMIT | PID_IMPROVE_DERIV_ON_MEAS
            | PID_IMPROVE_DERIV_FILTER;
    reset();
  }

  void reset() {
    I_term = 0.0f; previous_error = 0.0f;
    previous_input = 0.0f; previous_output = 0.0f;
    D_filtered = 0.0f;
    previous_time = 0; blocked_counter = 0;
    is_blocked = false;
  }

  float calculate(float setpoint, float input) {
    // micros() thay cho millis(): vòng điều khiển động cơ chạy 1 kHz thì
    // millis() có bước nhảy quá thô, dt bị lượng tử hóa tới 100%.
    unsigned long current_time = micros();
    if (previous_time == 0) {
      previous_time = current_time;
      previous_error = setpoint - input;
      previous_input = input;
      return 0.0f;
    }
    // Phép trừ số nguyên KHÔNG DẤU nên vẫn đúng khi micros() tràn 32-bit.
    float dt = (current_time - previous_time) * 1e-6f;
    if (dt <= 0.0f || dt > 0.5f) dt = 0.001f;
    previous_time = current_time;

    float error = setpoint - input;

    // 1. Chống kẹt động cơ — xét cả hai biên nên chạy được cả chiều lùi
    if (Improve & PID_IMPROVE_ERROR_HANDLE) {
      bool at_limit = (previous_output >= Max_Output) ||
                      (previous_output <= Min_Output);
      if (at_limit && fabs(input - previous_input) < 0.01f) {
        if (++blocked_counter > 30) {
          is_blocked = true;
          return 0.0f;
        }
      } else {
        blocked_counter = 0;
      }
    }

    // 2. Vùng chết: chỉ ngắt P và I. Khâu D VẪN chạy — vật đang trôi thì
    //    vẫn phải hãm. Và ta KHÔNG zero hóa biến error, xem giải thích dưới.
    bool in_deadband = (Improve & PID_IMPROVE_DEADBAND) &&
                       (fabs(error) < DeadBand);

    // 3. Khâu P
    float P_term = in_deadband ? 0.0f : Kp * error;

    // 4. Khâu D: vi phân theo giá trị đo + lọc thông thấp (toa thuốc 3, Bài 2)
    float D_raw = (Improve & PID_IMPROVE_DERIV_ON_MEAS)
                ? (Kd * (previous_input - input) / dt)
                : (Kd * (error - previous_error) / dt);
    D_filtered = (Improve & PID_IMPROVE_DERIV_FILTER)
               ? (Deriv_Alpha * D_raw + (1.0f - Deriv_Alpha) * D_filtered)
               : D_raw;
    float D_term = D_filtered;

    // 5. Khâu I: hình thang + biến tốc độ + KẸP CHỐNG WINDUP (toa thuốc 2, Bài 2)
    if (!in_deadband) {
      float factor = 1.0f;
      if (Improve & PID_IMPROVE_CHANGING_INT) {
        float abs_e = fabs(error);
        if (abs_e >= (ScalarA + ScalarB)) factor = 0.0f;
        else if (abs_e > ScalarB) factor = (ScalarA - abs_e + ScalarB) / ScalarA;
      }
      float I_step = (Improve & PID_IMPROVE_TRAPEZOID_INT)
                   ? (factor * Ki * 0.5f * (error + previous_error) * dt)
                   : (factor * Ki * error * dt);

      // Ngừng cộng nợ khi ngõ ra đã chạm biên mà bước này còn đẩy sâu thêm
      float tentative = P_term + I_term + I_step + D_term;
      bool pushing_into_limit =
            (tentative > Max_Output && I_step > 0.0f) ||
            (tentative < Min_Output && I_step < 0.0f);
      if (!pushing_into_limit) I_term += I_step;

      if (Improve & PID_IMPROVE_INTEGRAL_LIMIT) {
        I_term = constrain(I_term, -Max_I, Max_I);
      }
    }

    // 6. Tổng ngõ ra có lọc và bão hòa — dải HAI CHIỀU
    float output = constrain(P_term + I_term + D_term, Min_Output, Max_Output);
    if (Improve & PID_IMPROVE_OUTPUT_FILTER) {
      output = Output_Alpha * output + (1.0f - Output_Alpha) * previous_output;
      output = constrain(output, Min_Output, Max_Output);
    }

    previous_error = error;   // lưu sai số THÔ, không phải bản đã zero hóa
    previous_input = input;
    previous_output = output;
    return output;
  }

private:
  float I_term, previous_error, previous_input, previous_output, D_filtered;
  unsigned long previous_time;
  uint16_t blocked_counter;
};

    Năm chi tiết dễ sai mà bản code trên đã xử lý

    a. Ngõ ra phải hai chiều. Một chiếc xe bốn bánh hay gimbal cần quay được cả hai hướng, nên biên dưới là Min_Output (âm) chứ không phải 0.0f. Kẹp sàn ở 0 là bộ điều khiển một chiều — chỉ đúng cho lò sấy.

    b. Kẹp chống windup phải còn nguyên. Toa thuốc 2 của Bài 2 ngừng cộng nợ khi đã bão hòa. Nếu đóng gói mà chỉ giữ lại constrain(I_term, ...) thì khâu I vẫn phình tự do tới trần rồi mới bị chặn — yếu hơn hẳn.

    c. Lọc khâu D phải còn nguyên. Tương tự, toa thuốc 3 là thứ bắt buộc với cảm biến thật. Cờ PID_IMPROVE_DERIV_FILTER giữ nó lại trong thư viện.

    d. micros() chứ không phải millis(). Vòng điều khiển động cơ chạy 1 kHz thì millis() cho dt chỉ có hai giá trị 0 hoặc 1 ms — sai số lượng tử hóa 100%. Đây chính là điều Bài 5 sẽ chỉ ra trong mã nguồn SimpleFOC.

    e. Đừng zero hóa biến error khi vào vùng chết. Cách viết if (|e| < DeadBand) error = 0; rất tự nhiên nhưng có một tác dụng phụ ngầm: previous_error lưu lại con số 0 đó, nên ngay nhịp sau khâu D tính ra một cú vi phân giả từ bước nhảy nhân tạo $e: 0.3 \to 0$. Cách đúng là giữ nguyên error và chỉ ngắt đóng góp của P và I bằng cờ in_deadband.

    Bạn muốn bản đã hoàn thiện? Toàn bộ những điểm trên, cộng thêm feedforward, setpoint weighting và chống windup kiểu back-calculation, đã được gói sẵn trong shark_pid — thư viện mã nguồn mở tôi viết làm phần kết cho series này. Lõi C99 thuần chạy được cả trên STM32 lẫn Arduino, kèm bộ test tự động.

    4. Ứng dụng điều khiển nhiều đối tượng cùng lúc

    Nhờ đóng gói OOP, việc điều khiển đồng thời nhiều cơ cấu trong loop() trở nên cực kỳ trong sáng:

    // Khởi tạo 2 đối tượng PID độc lập.
// Dải ngõ ra -255..255: bánh xe phải quay được cả hai chiều.
PIDController motorLeft(3.0, 0.2, 1.0, -255.0, 255.0, 30.0);
PIDController motorRight(3.0, 0.2, 1.0, -255.0, 255.0, 30.0);

void setup() {
  // Kích hoạt trọn bộ tính năng tối ưu cho cả 2 motor
  uint8_t fullFlags = PID_IMPROVE_INTEGRAL_LIMIT | 
                      PID_IMPROVE_DERIV_ON_MEAS  | 
                      PID_IMPROVE_DERIV_FILTER   | 
                      PID_IMPROVE_TRAPEZOID_INT  | 
                      PID_IMPROVE_OUTPUT_FILTER  | 
                      PID_IMPROVE_DEADBAND       | 
                      PID_IMPROVE_ERROR_HANDLE;
  motorLeft.Improve = fullFlags;
  motorRight.Improve = fullFlags;
}

void loop() {
  float speedL = readSpeedLeft();
  float speedR = readSpeedRight();

  float pwmLeft  = motorLeft.calculate(100.0, speedL);
  float pwmRight = motorRight.calculate(100.0, speedR);

  setMotorPWM(pwmLeft, pwmRight);
}

    6. Tài liệu tham khảo & Mã nguồn gốc

    • Kho lưu trữ GitHub: WangHongxi2001/PID_Library — Thư viện PID của WangHongxi với kiến trúc cờ Bitmask Improve, bộ lọc ngõ ra và cơ chế ngắt chống kẹt động cơ. Ý tưởng thuật toán rất đáng học; nhưng bản trên GitHub dừng ở v1.0.6 từ 2020 và có vài chỗ hở đáng biết trước khi copy về dùng — chi tiết ở bảng đối chiếu này.

    • shark_pid — thư viện tôi viết làm phần kết cho series. Lõi C99 không phụ thuộc Arduino/HAL nên dùng chung được cho STM32, ESP-IDF và Arduino; dt là tham số tường minh nên đổi tần số vòng lặp không phải dò lại hệ số. Có sẵn bộ test tự động và CI.

    Đọc mã nguồn phải biết phản biện. Thư viện mã nguồn mở nổi tiếng không đồng nghĩa với không có lỗi. Ngay trong PID.h của bản gốc có dòng #define ABS(x) ((x > 0) ? x : -x) — thiếu cặp ngoặc quanh x, nên ABS(a - b) nở ra thành -a - b chứ không phải |a - b|, sai hẳn dấu khi b > a. Macro đó được dùng ngay trong hàm phát hiện kẹt động cơ. Kỹ năng đáng giá nhất mà series này muốn trao cho bạn không phải là copy được code, mà là đọc được code và nhận ra chỗ nào đáng ngờ.
